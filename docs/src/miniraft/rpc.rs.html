6/// A message can be either targeted at a single server or to everyone
7pub type SendableMessage<T> = (Target, RPC<T>);
8
9/// Whether to send a message to everyone or just a single node
10#[derive(Eq, PartialEq, PartialOrd, Ord, Clone, Copy)]
11pub enum Target {
12    /// A single server
13    Single(ServerId),
14    /// To everyone
15    Broadcast,
16}
17
18/// A Raft RPC request
19pub enum RPC<T> {
20    /// Candidate requesting to become leader
21    VoteRequest(VoteRequest),
22    /// Response to [`VoteRequest`]
23    VoteResponse(VoteResponse),
24    /// Leader heartbeat/appending entries to followers
25    AppendRequest(AppendRequest<T>),
26    /// Response to [`AppendRequest`]
27    AppendResponse(AppendResponse),
28}
29
30/// Request by a candidate to become a Raft leader
31pub struct VoteRequest {
32    /// Current term of candidate
33    pub candidate_term: Term,
34    /// ID of candidate requesting a vote
35    pub candidate_id: ServerId,
36    /// Index of candidate's last log entry
37    pub candidate_last_log_idx: LogIndex,
38    /// Term of candidate's last log entry
39    pub candidate_last_log_term: Term,
40}
41
42/// Response to a [`VoteRequest`]
43pub struct VoteResponse {
44    /// [`current_term`](RaftServer::current_term) of server for candidate to update itself
45    pub term: Term,
46    /// Whether the [`VoteRequest`] was granted or not
47    pub vote_granted: bool,
48    /// Who sent the vote
49    pub votee_id: ServerId,
50}
51
52/// Request from leader to append entries to follower's log
53#[derive(Clone)]
54pub struct AppendRequest<T> {
55    /// Term of leader requesting log append
56    pub leader_term: Term,
57    /// ID of leader (used so follower can redirect clients)
58    pub leader_id: ServerId,
59    /// Log index immediately preceding index of next element in [`entries`](Self::entries)
60    pub leader_last_log_idx: LogIndex,
61    /// Term of [`leader_last_log_idx`](Self::leader_last_log_idx)
62    pub leader_last_log_term: Term,
63    /// Leader's [`committed_len`](Log::committed_len)
64    pub leader_commit: LogIndex,
65    /// A list of consecutive log entries to append to follower
66    pub entries: Vec<LogEntry<T>>,
67}
68
69/// Response to an [`AppendRequest`]
70pub struct AppendResponse {
71    /// Whether the follower added it to their log or not
72    pub ok: bool,
73    /// [`current_term`](RaftServer::current_term) of server for candidate to update itself
74    pub term: Term,
75    /// Index of the last log entry we appended to the log
76    pub ack_idx: LogIndex,
77    /// Follower ID
78    pub follower_id: ServerId,
79}

81/// Display trait implementations
82impl<T> Display for RPC<T> {
83    fn fmt(&self, f: &mut Formatter) -> Result {
84        write!(
85            f,
86            "{}",
87            match self {
88                RPC::VoteRequest(_) => "VoteRequest",
89                RPC::AppendRequest(_) => "AppendRequest",
90                RPC::VoteResponse(_) => "VoteResponse",
91                RPC::AppendResponse(_) => "AppendResponse",
92            }
93        )
94    }
95}
